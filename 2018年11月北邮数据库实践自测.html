create database kaoshi;
use kaoshi;
# 1
create table customer(
	mid char(5) comment '成员ID（主键）' primary key,
	nam varchar(20) not null default '' comment '姓名',
	birth date comment '生日',
	sex char(6) not null default '' comment '性别，女：1 男：0'
) comment '成员表';
create table users(
	uid char(6) comment '用户id（主键）' primary key,
	zip char(8) not null default '' comment '邮编',
	address varchar(255) not null default '' comment '住址',
	`name` varchar(20) not null default '' comment '姓名'
) comment '用户表';
create table order_basic(
	oid char(15) comment '订单id(主键)' primary key,
	odate date comment '发单日期',
	uid char(6) comment '用户id' ,
	memo varchar(255) not null default '' comment '备注',
	foreign key(uid) references users(uid)
) comment '订单表';
create table order_details(
	oid char(15) comment '订单id',
	pid char(9) comment '产品id',
	quantity int not null default 0 comment '数量',
	primary key (oid, pid)
) comment '订单详细表';
create table product(
	pid char(9) comment '产品id(主键)' primary key,
	pname varchar(255) not null default '' comment '产品名',
	price float(10, 1) not null default 0 comment '价格'
) comment '产品表';

create view order_view 
as 
select ob.oid, ob.uid, u.name, p.pname, p.price, od.quantity 
from users u
inner join order_basic ob on ob.uid = u.uid
inner join order_details od on ob.oid = od.oid 
inner join product p on od.pid = p.pid;
